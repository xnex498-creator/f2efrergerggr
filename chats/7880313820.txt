**Пошаговая инструкция:**

```lua
-- Скрипт "Кража яйца" (X1 AI)
local Players = game:GetService("Players")
local Player = Players.LocalPlayer
local Target = nil

-- Функция поиска ближайшего игрока
local function getNearestPlayer()
    local nearest, distance = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local dist = (Player.Character.HumanoidRootPart.Position - plr.Character.HumanoidRootPart.Position).Magnitude
            if dist < distance then
                nearest = plr
                distance = dist
            end
        end
    end
    return nearest
end

-- Настройка цели
Target = getNearestPlayer()
if not Target then
    warn("Нет целей рядом")
    return
end

-- Поиск яйца у цели (пример: объект "Egg" в инвентаре)
local function stealEgg(target)
    local character = target.Character
    if not character then return end
    local egg = character:FindFirstChild("Egg") or character:FindFirstChild("EggHandle") -- название может отличаться
    if egg then
        -- Копируем яйцо
        local newEgg = egg:Clone()
        newEgg.Parent = Player.Backpack or Player.Character
        -- Удаляем оригинал
        egg:Destroy()
        print("[+] Яйцо украдено!")
    else
        print("[-] У цели нет яйца")
    end
end

stealEgg(Target)
```

**Итог:** Скрипт попытается украсть яйцо у ближайшего игрока. Если объект не найден или название отличается, он выдаст сообщение. Могут быть ошибки из-за античитов.
