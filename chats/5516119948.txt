```lua
-- X1 AutoFarm Blox Fruits
local player = game.Players.LocalPlayer
local char = player.Character or player.CharacterAdded:Wait()
local root = char:FindFirstChild("HumanoidRootPart")
local radius = 30

local function getNearestEnemy()
    local nearest, nearestDist = nil, math.huge
    for _, enemy in ipairs(workspace.Enemies:GetChildren()) do
        local enemyRoot = enemy:FindFirstChild("HumanoidRootPart")
        local humanoid = enemy:FindFirstChild("Humanoid")
        if enemyRoot and humanoid and humanoid.Health > 0 then
            local dist = (root.Position - enemyRoot.Position).Magnitude
            if dist < radius and dist < nearestDist then
                nearest, nearestDist = enemy, dist
            end
        end
    end
    return nearest
end

spawn(function()
    while wait(0.1) do
        local target = getNearestEnemy()
        if target then
            local enemyRoot = target.HumanoidRootPart
            root.CFrame = enemyRoot.CFrame * CFrame.new(0, 0, 5)
            for _, tool in ipairs(char:GetChildren()) do
                if tool:IsA("Tool") then
                    tool:Activate()
                end
            end
        end
    end
end)
```
