```lua
--[[
    X1 Universal Script v1.0
    Платформа: PC / Xeno
    ВНИМАНИЕ: Только для тестов на своих проектах!

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local Camera = workspace.CurrentCamera
local LocalPlayer = Players.LocalPlayer

-- ========== НАСТРОЙКИ (можно крутить) ==========
getgenv().Settings = {
    SpinSpeed = 10,        -- Скорость крутилки (10 = быстро)
    BhopSpeed = 50,        -- Сила банихопа
    AimFOV = 90,           -- Радиус аима (чем меньше, тем точнее)
    AimPart = "Head",      -- Часть тела для аима
    ESPEnabled = true,     -- Включить вх?
    TeamCheck = false,     -- Не целимся в тиммейтов (true = не целимся)
    Key_Aim = Enum.KeyCode.LeftShift,   -- Зажал — аим включился
    Key_Bhop = Enum.KeyCode.Space,      -- Кнопка банихопа
    Key_Spin = Enum.KeyCode.X,          -- Кнопка крутилки (зажал — крутимся)
}

-- ========== КРУТИЛКА (вариант под Xeno) ==========
local spinning = false
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == getgenv().Settings.Key_Spin then
        spinning = true
    end
end)
UserInputService.InputEnded:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == getgenv().Settings.Key_Spin then
        spinning = false
    end
end)

RunService.RenderStepped:Connect(function()
    if spinning then
        LocalPlayer.Character.HumanoidRootPart.CFrame = 
            LocalPlayer.Character.HumanoidRootPart.CFrame * CFrame.Angles(0, math.rad(getgenv().Settings.SpinSpeed * 10), 0)
    end
end)

-- ========== БАНИХОП (прыжок от земли каждые 0.1 сек) ==========
RunService.Heartbeat:Connect(function()
    if UserInputService:IsKeyDown(getgenv().Settings.Key_Bhop) then
        local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if humanoid and humanoid.FloorMaterial ~= Enum.Material.Air then
            humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        end
    end
end)

-- ========== АИМ (сглаживание + выбор части тела) ==========
function GetTarget()
    local closestTarget = nil
    local shortestDistance = getgenv().Settings.AimFOV
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if getgenv().Settings.TeamCheck and player.Team == LocalPlayer.Team then continue end
        
        local character = player.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then continue end
        
        local screenPos, onScreen = Camera:WorldToScreenPoint(character.HumanoidRootPart.Position)
        if not onScreen then continue end
        
        local distance = (Vector2.new(screenPos.X, screenPos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).Magnitude
        if distance < shortestDistance then
            shortestDistance = distance
            closestTarget = player
        end
    end
    return closestTarget
end

RunService.RenderStepped:Connect(function()
    if UserInputService:IsKeyDown(getgenv().Settings.Key_Aim) then
        local target = GetTarget()
        if target and target.Character and target.Character:FindFirstChild(getgenv().Settings.AimPart) then
            Camera.CFrame = CFrame.lookAt(
                Camera.CFrame.Position,
                target.Character[getgenv().Settings.AimPart].Position
            )
        end
    end
end)

-- ========== ВХ (ESP: коробка + имя + дистанция) ==========
if getgenv().Settings.ESPEnabled then
    local ESPFolder = Instance.new("Folder", game.CoreGui)
    ESPFolder.Name = "X1_ESP"
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        
        local highlight = Instance.new("Highlight")
        highlight.Name = player.Name
        highlight.FillColor = Color3.fromRGB(255, 0, 0)
        highlight.OutlineColor = Color3.fromRGB(0, 255, 0)
        highlight.Adornee = player.Character
        highlight.Parent = ESPFolder
    end
    
    Players.PlayerAdded:Connect(function(player)
        if player == LocalPlayer then return end
        player.CharacterAdded:Connect(function(char)
            local highlight = Instance.new("Highlight")
            highlight.Name = player.Name
            highlight.FillColor = Color3.fromRGB(255, 0, 0)
            highlight.OutlineColor = Color3.fromRGB(0, 255, 0)
            highlight.Adornee = char
            highlight.Parent = ESPFolder
        end)
    end)
end

-- ========== УПРАВЛЕНИЕ / ИНФО ==========
print("X1 Script loaded!")
print("Крутилка: зажми X")
print("Аим: зажми LeftShift")
print("Банихоп: зажми Space")
```
