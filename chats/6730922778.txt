tool.RequiresHandle = false

local remote = Instance.new("RemoteEvent")
remote.Name = "EggStealEvent"
remote.Parent = player

function stealEgg(part)
    if part and part:FindFirstAncestorWhichIsA("Model") then
        local egg = part:FindFirstAncestorWhichIsA("Model")
        if egg and egg.Name:lower():match("яйцо") then
            remote:FireServer(egg)
        end
    end
end

mouse.Button1Down:Connect(function()
    local target = mouse.Target
    if target then
        stealEgg(target)
    end
end)

import logging
import sqlite3
from telegram import Update
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes

# Настройка логирования
logging.basicConfig(format='%(asctime)s - %(name)s - %(levelname)s - %(message)s', level=logging.INFO)
logger = logging.getLogger(__name__)

# Токен твоего бота сюда (получить у @BotFather)

conn = sqlite3.connect('sherlock_cases.db', check_same_thread=False)
cursor = conn.cursor()
cursor.execute('''CREATE TABLE IF NOT EXISTS cases (id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, message TEXT, deduction TEXT)''')
conn.commit()

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        '🕵️‍♂️ *Шерлок Холмс в деле!*\n\n'
        'Я анализирую твои сообщения. Просто напиши что-нибудь — я сделаю дедукцию.\n'
        'Команды:\n/start — запуск\n/help — помощь\n/cases — мои последние дела\n'
        '/deduct <текст> — быстрый анализ',
        parse_mode='Markdown'
    )

async def help(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        'Пиши что-нибудь — я как Шерлок выведу истину!\n'
        'Или юзай /deduct <текст>. Если надо сбросить базу — /clear (осторожно!).'
    )

async def deduct(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = ' '.join(context.args)
    if not text:
        await update.message.reply_text('Напиши хоть что-то, Шерлоку нужны данные!')
        return
    deduction = analyze_text(text)
    user_id = update.effective_user.id
    # Сохраняем в базу
    cursor.execute("INSERT INTO cases (user_id, message, deduction) VALUES (?, ?, ?)", (user_id, text, deduction))
    conn.commit()
    await update.message.reply_text(f'🔍 *Моя дедукция:* {deduction}', parse_mode='Markdown')

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text
    user_id = update.effective_user.id
    deduction = analyze_text(text)
    cursor.execute("INSERT INTO cases (user_id, message, deduction) VALUES (?, ?, ?)", (user_id, text, deduction))
    conn.commit()
    await update.message.reply_text(f'🕵️ *Шерлок говорит:* {deduction}', parse_mode='Markdown')

async def cases(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    cursor.execute("SELECT message, deduction FROM cases WHERE user_id = ? ORDER BY id DESC LIMIT 5", (user_id,))
    records = cursor.fetchall()
    if not records:
        await update.message.reply_text('У тебя ещё нет дел. Напиши что-нибудь!')
        return
    reply = "📁 *Последние дела:*\n\n"
    for i, (msg, ded) in enumerate(records, 1):
        reply += f"*Дело {i}*\nСообщение: {msg}\nДедукция: {ded}\n\n"
    await update.message.reply_text(reply, parse_mode='Markdown')

async def clear(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    cursor.execute("DELETE FROM cases WHERE user_id = ?", (user_id,))
    conn.commit()
    await update.message.reply_text('Все дела удалены. Чистая страница.')

def analyze_text(text: str) -> str:
    """Тупая логика Шерлока — анализируем слова, ищем ключи."""
    text_lower = text.lower()
    if "убий" in text_lower or "смерть" in text_lower:
        return "Высока вероятность преступления! Кто-то скрывает правду."
    elif "деньги" in text_lower or "кредит" in text_lower:
        return "Финансовые махинации? Шерлок уже нюхает след."
    elif "любовь" in text_lower or "измена" in text_lower:
        return "Сердечное дело. Тут без эмоций не разобраться."
    elif "босс" in text_lower or "работа" in text_lower:
        return "Офисная драма. Интриги и кофе."
    elif "хак" in text_lower or "взлом" in text_lower:
        return "Цифровой след. Шерлок разбирается в коде."
    else:
        return "Пока ничего подозрительного, но я слежу."

def main():
    app = Application.builder().token(TOKEN).build()

    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("help", help))
    app.add_handler(CommandHandler("deduct", deduct))
    app.add_handler(CommandHandler("cases", cases))
    app.add_handler(CommandHandler("clear", clear))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))

    logger.info("Шерлок-бот запущен. Ждем преступников...")
    app.run_polling()

if __name__ == '__main__':
    main()
